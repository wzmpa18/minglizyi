## 一、总体规范

### 1.1 调用原则

1. 所有算法函数均为纯函数，无副作用，可在服务端和客户端直接调用
2. Next.js 服务端（SSR/SSG）与客户端渲染共用同一套算法代码，保证两端计算结果完全一致
3. 算法函数不依赖任何 UI 框架或浏览器 API
4. 所有入参必须经过校验，异常数据返回友好错误提示

### 1.2 通用类型定义

```typescript
// 天干
type TianGan = '甲' | '乙' | '丙' | '丁' | '戊' | '己' | '庚' | '辛' | '壬' | '癸';

// 地支
type DiZhi = '子' | '丑' | '寅' | '卯' | '辰' | '巳' | '午' | '未' | '申' | '酉' | '戌' | '亥';

// 五行
type WuXing = '木' | '火' | '土' | '金' | '水';

// 阴阳
type YinYang = '阳' | '阴';

// 性别
type Gender = 'male' | 'female';

// 十神
type ShiShen = '比肩' | '劫财' | '食神' | '伤官' | '偏财' | '正财' | '七杀' | '正官' | '偏印' | '正印';

// 十二长生阶段
type ShengWangStage = '长生' | '沐浴' | '冠带' | '临官' | '帝旺' | '衰' | '病' | '死' | '墓' | '绝' | '胎' | '养';
```

### 1.3 错误码规范

| 错误码 | 说明 | 处理方式 |
|--------|------|----------|
| `E001` | 无效天干 | 提示用户输入正确的天干 |
| `E002` | 无效地支 | 提示用户输入正确的地支 |
| `E003` | 无效干支组合 | 提示用户输入正确的干支 |
| `E004` | 无效出生日期 | 提示用户输入正确的日期 |
| `E005` | 无效出生时间 | 提示用户输入正确的时间 |
| `E006` | 算法内部错误 | 联系技术支持 |

## 二、八字模块接口

### 2.1 八字排盘

```typescript
/**
 * 八字排盘完整计算
 * @param year   出生年份
 * @param month  出生月份 (1-12)
 * @param day    出生日期
 * @param hour   出生小时 (0-23)
 * @param minute 出生分钟 (0-59)
 * @param gender 性别
 * @returns BaziResult
 */
function calculateBazi(
  year: number,
  month: number,
  day: number,
  hour: number,
  minute: number,
  gender: Gender
): BaziResult;

interface BaziResult {
  yearPillar:  { gan: TianGan; zhi: DiZhi; ganzhi: string; nayin: string; wuxing: WuXing };
  monthPillar: { gan: TianGan; zhi: DiZhi; ganzhi: string; nayin: string; wuxing: WuXing };
  dayPillar:   { gan: TianGan; zhi: DiZhi; ganzhi: string; nayin: string; wuxing: WuXing };
  hourPillar:  { gan: TianGan; zhi: DiZhi; ganzhi: string; nayin: string; wuxing: WuXing };
  dayGan: TianGan;           // 日干
  cangGan: Record<DiZhi, TianGan[]>;  // 藏干
  shiShen: Record<string, ShiShen>;    // 十神
  kongWang: DiZhi[];                   // 空亡
  changSheng: Record<string, ShengWangStage>; // 十二长生
  shenQiangRuo: ShenQiangRuoResult;    // 身强身弱
  dayun: DayunResult;                  // 大运
  pattern: PatternResult;              // 格局
}
```

### 2.2 身强身弱判定

```typescript
interface ShenQiangRuoResult {
```

### 2.3 大运计算

```typescript
interface DayunResult {
  forward: boolean;        // 顺排/逆排
  direction: string;       // '顺排' | '逆排'
  startAge: number;        // 起运年龄（精确到小数点后2位）
  startDate: string;       // 起运日期 'YYYY-MM-DD'
  startYear: number;       // 起运年份
  dayunList: DayunItem[];

interface DayunItem {
  ganzhi: string;          // 大运干支
  order: number;           // 第几运
  startAge: number;        // 起运年龄
  startYear: number;       // 起运年份
}
```

### 2.4 格局判定

```typescript
interface PatternResult {
  patterns: string[];      // 格局列表
  mainPattern: string;     // 主格局
  patternType: string;     // '正八格' | '特殊格局'
  detail: {
    monthZhi: DiZhi;
    cangGan: TianGan[];
    benQi: TianGan;
    benQiShiShen: ShiShen;
```

```typescript
function validateBaziInput(year: number, month: number, day: number, hour: number): void {
  if (year < 1900 || year > 2100) throw new Error('E004: 出生年份超出范围');
  if (month < 1 || month > 12) throw new Error('E004: 出生月份无效');
  if (day < 1 || day > 31) throw new Error('E004: 出生日期无效');
  if (hour < 0 || hour > 23) throw new Error('E005: 出生时间无效');
}
```
