export interface ConsumptionRebateResult {
  granted: boolean;
  reason?: string;
  level1Points?: number;
  level2Points?: number;
}

export async function reportConsumptionRebate(params: {
  orderNo: string;
  amount: number;
  product?: string;
}): Promise<ConsumptionRebateResult | null> {
  try {
    const res = await fetch(`${API_BASE}/api/auth/invite/consumption-rebate`, {
      method: "POST",
      headers: { "Content-Type": "application/json", ...tokenHeader() },
      body: JSON.stringify(params),
    });
    if (!res.ok) return null;
    const data = await res.json();
    if (!data || !data.success || !data.data) return null;
    return data.data as ConsumptionRebateResult;
  } catch {
    return null;
  }
}
